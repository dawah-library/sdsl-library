    document.addEventListener('DOMContentLoaded', function() {
        if (typeof EventFactory === 'undefined') return;

        // PAGE_VIEW — dedup: skip if same page was already tracked in this tab session
        var pvKey = 'last_tracked_page_view';
        var currentPage = window.location.pathname + window.location.search;
        if (sessionStorage.getItem(pvKey) !== currentPage) {
            EventFactory.create('PAGE_VIEW', {
                page_url: window.location.href,
                page_title: document.title
            });
            sessionStorage.setItem(pvKey, currentPage);
        }

    function showNotationTooltipDiv(notationValue) {
        const popoverContent = document.querySelector(".popover-content");
